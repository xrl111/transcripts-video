# Local Speech-to-Text Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Công cụ dòng lệnh `stt` chạy local trên Windows: thu âm thanh máy tính hoặc nạp video, chép lời bằng Whisper, làm sạch bằng quy tắc, xuất `transcript.md`.

**Architecture:** Clean Architecture bốn tầng trong `src/stt/`: `domain` (entity và quy tắc làm sạch thuần), `application` (DTO, ports, use case), `infrastructure` (adapter cho faster-whisper, WASAPI, JSONL, Markdown), `presentation` (CLI, nơi lắp ráp). Mỗi bước ghi kết quả ra file trong `out/<ngày>-<tên>/`.

**Tech Stack:** Python 3.11, faster-whisper, PyAudioWPatch, pytest, thư viện chuẩn (`dataclasses`, `argparse`, `wave`, `json`).

**Spec:** `docs/superpowers/specs/2026-10-09-local-stt-design.md`

## Global Constraints

- Python 3.11 trên Windows 11, chỉ CPU: `device="cpu"`, `compute_type="int8"`.
- `domain` chỉ import thư viện chuẩn. `application` chỉ import `domain` và thư viện chuẩn. Chỉ `infrastructure` import `faster_whisper` và `pyaudiowpatch`.
- Phụ thuộc: chỉ `faster-whisper`, `PyAudioWPatch`, và `pytest` cho dev. Không thêm thư viện khác.
- DTO và entity là `@dataclass(frozen=True)`.
- Mọi đọc/ghi file văn bản dùng `encoding="utf-8"`.
- Lỗi dành cho người dùng là `SttError` với thông báo tiếng Việt; chỉ `cli.py` được `print`.
- Test không được cần model, mạng hay thiết bị âm thanh.
- Lệnh chạy test: `.venv/Scripts/python -m pytest -q`

`src/stt/domain/cleaning.py`:

```python
from dataclasses import dataclass, replace
from typing import Iterable

from stt.domain.entities import Paragraph, Segment


@dataclass(frozen=True)
class CleaningRules:
    max_no_speech_prob: float = 0.6
    min_avg_logprob: float = -1.0
    max_compression_ratio: float = 2.4
    paragraph_pause: float = 2.0
    paragraph_max_words: int = 150


def _is_speech(segment: Segment, rules: CleaningRules) -> bool:
    silent = (
        segment.no_speech_prob > rules.max_no_speech_prob
        and segment.avg_logprob < rules.min_avg_logprob
    )
    return not silent and segment.compression_ratio <= rules.max_compression_ratio


def filter_segments(segments: Iterable[Segment], rules: CleaningRules) -> list[Segment]:
    kept: list[Segment] = []
    for segment in segments:
        text = " ".join(segment.text.split())
        if not text or not _is_speech(segment, rules):
            continue
        if kept and kept[-1].text.casefold() == text.casefold():
            continue
        kept.append(replace(segment, text=text))
    return kept


def to_paragraphs(segments: Iterable[Segment], rules: CleaningRules) -> list[Paragraph]:
    paragraphs: list[Paragraph] = []
    words: list[str] = []
    start = last_end = 0.0
    for segment in segments:
        too_long = len(words) >= rules.paragraph_max_words
        if words and (segment.start - last_end > rules.paragraph_pause or too_long):
            paragraphs.append(Paragraph(start, " ".join(words)))
            words = []
        if not words:
            start = segment.start
        words.extend(segment.text.split())
        last_end = segment.end
    if words:
        paragraphs.append(Paragraph(start, " ".join(words)))
    return paragraphs


def clean(segments: Iterable[Segment], rules: CleaningRules = CleaningRules()) -> list[Paragraph]:
    return to_paragraphs(filter_segments(segments, rules), rules)
```

```python
from dataclasses import dataclass
from pathlib import Path
from typing import BinaryIO, Iterator, Protocol

from stt.domain.entities import Paragraph, Segment, TranscriptMeta


class SttError(Exception):
    """Lỗi có thông báo dành cho người dùng."""


@dataclass(frozen=True)
class Recognition:
    language: str
    duration: float
    segments: Iterator[Segment]


class AudioRecorder(Protocol):
    def record(self, output_path: Path) -> Iterator[bytes]: ...


class SpeechRecognizer(Protocol):
    model_name: str

    def transcribe(
        self, source: Path | BinaryIO, language: str | None, prompt: str | None
    ) -> Recognition: ...


class SegmentWriter(Protocol):
    def write(self, segment: Segment) -> None: ...

    def close(self, complete: bool) -> None: ...


class SegmentStore(Protocol):
    def open_writer(self, path: Path, meta: TranscriptMeta) -> SegmentWriter: ...

    def read(self, path: Path) -> tuple[TranscriptMeta, list[Segment], bool]: ...


class TranscriptWriter(Protocol):
    def write(self, path: Path, meta: TranscriptMeta, paragraphs: list[Paragraph]) -> None: ...
```

```python
from datetime import date
from typing import Callable

from stt.application.dto import CleanRequest, CleanResult, TranscribeRequest, TranscribeResult
from stt.application.ports import SegmentStore, SpeechRecognizer, SttError, TranscriptWriter
from stt.domain.cleaning import filter_segments, to_paragraphs
from stt.domain.entities import TranscriptMeta


class TranscribeMedia:
    def __init__(self, recognizer: SpeechRecognizer, store: SegmentStore):
        self._recognizer = recognizer
        self._store = store

    def execute(
        self,
        request: TranscribeRequest,
        on_progress: Callable[[float, float], None] = lambda done, total: None,
    ) -> TranscribeResult:
        if not request.source.is_file():
            raise SttError(f"Không tìm thấy file: {request.source}")
        raw_path = request.out_dir / "raw.jsonl"
        if raw_path.exists():
            raise SttError(f"{raw_path} đã tồn tại. Xóa hoặc đổi tên nó trước khi chép lại.")
        request.out_dir.mkdir(parents=True, exist_ok=True)

        recognition = self._recognizer.transcribe(request.source, request.language, request.prompt)
        meta = TranscriptMeta(
            source=request.source.name,
            created=date.today().isoformat(),
            duration=recognition.duration,
            language=recognition.language,
            model=self._recognizer.model_name,
        )
        writer = self._store.open_writer(raw_path, meta)
        count = 0
        complete = False
        try:
            for segment in recognition.segments:
                writer.write(segment)
                count += 1
                on_progress(segment.end, recognition.duration)
            complete = True
        finally:
            writer.close(complete)
        return TranscribeResult(raw_path, recognition.language, recognition.duration, count)


class CleanTranscript:
    def __init__(self, store: SegmentStore, writer: TranscriptWriter):
        self._store = store
        self._writer = writer

    def execute(self, request: CleanRequest) -> CleanResult:
        if not request.raw_path.is_file():
            raise SttError(f"Không tìm thấy file: {request.raw_path}")
        meta, segments, complete = self._store.read(request.raw_path)
        kept = filter_segments(segments, request.rules)
        transcript_path = request.raw_path.with_name("transcript.md")
        self._writer.write(transcript_path, meta, to_paragraphs(kept, request.rules))
        return CleanResult(transcript_path, len(kept), len(segments) - len(kept), complete)
```

```python
from pathlib import Path
from typing import TextIO

from stt.application.ports import SttError
from stt.domain.entities import Segment, TranscriptMeta


class _JsonlWriter:
    def __init__(self, file: TextIO):
        self._file = file

    def write(self, segment: Segment) -> None:
        self.write_row({"type": "segment", **asdict(segment)})

    def close(self, complete: bool) -> None:
        if complete:
            self.write_row({"type": "end"})
        self._file.close()

    def write_row(self, row: dict) -> None:
        self._file.write(json.dumps(row, ensure_ascii=False) + "\n")
        self._file.flush()


class JsonlStore:
    def open_writer(self, path: Path, meta: TranscriptMeta) -> _JsonlWriter:
        writer = _JsonlWriter(path.open("w", encoding="utf-8"))
        writer.write_row({"type": "meta", **asdict(meta)})
        return writer

    def read(self, path: Path) -> tuple[TranscriptMeta, list[Segment], bool]:
        meta = None
        segments: list[Segment] = []
        complete = False
        try:
            with path.open(encoding="utf-8") as file:
                for line in file:
                    try:
                        row = json.loads(line)
                    except json.JSONDecodeError:
                        break  # dòng cuối bị cắt dở khi tiến trình chết giữa chừng
                    kind = row.pop("type", None)
                    if kind == "meta":
                        meta = TranscriptMeta(**row)
                    elif kind == "segment":
                        segments.append(Segment(**row))
                    elif kind == "end":
                        complete = True
        except (TypeError, AttributeError, UnicodeDecodeError) as error:
            raise SttError(f"{path} không đúng định dạng raw.jsonl: {error}") from error
        if meta is None:
            raise SttError(f"{path} không đúng định dạng raw.jsonl: thiếu dòng meta")
        return meta, segments, complete
```

```python
import json
from pathlib import Path

from stt.domain.entities import Paragraph, TranscriptMeta


def _stamp(seconds: float) -> str:
    total = int(seconds)
    return f"{total // 3600:02d}:{total % 3600 // 60:02d}:{total % 60:02d}"


class MarkdownWriter:
    def write(self, path: Path, meta: TranscriptMeta, paragraphs: list[Paragraph]) -> None:
        lines = [
            "---",
            f"source: {json.dumps(meta.source, ensure_ascii=False)}",
            f"created: {meta.created}",
            f'duration: "{_stamp(meta.duration)}"',
            f"language: {meta.language}",
            f"model: {meta.model}",
            "---",
            "",
        ]
        for paragraph in paragraphs:
            lines += [f"[{_stamp(paragraph.start)}] {paragraph.text}", ""]
        path.write_text("\n".join(lines), encoding="utf-8")
```

```python
from pathlib import Path
from typing import BinaryIO

from faster_whisper import WhisperModel

from stt.application.ports import Recognition, SttError
from stt.domain.entities import Segment


class WhisperRecognizer:
    def __init__(self, model_name: str):
        self.model_name = model_name
        self._model: WhisperModel | None = None

    def transcribe(
        self, source: Path | BinaryIO, language: str | None, prompt: str | None
    ) -> Recognition:
        if self._model is None:
            try:
                self._model = WhisperModel(self.model_name, device="cpu", compute_type="int8")
            except Exception as error:
                raise SttError(f"Không nạp được model {self.model_name}: {error}") from error
        try:
            segments, info = self._model.transcribe(
                str(source) if isinstance(source, Path) else source,
                language=language,
                initial_prompt=prompt,
                vad_filter=True,
                beam_size=5,
                condition_on_previous_text=False,
            )
        except Exception as error:
            raise SttError(f"Không đọc được âm thanh từ nguồn: {error}") from error
        return Recognition(
            info.language,
            info.duration,
            (
                Segment(s.start, s.end, s.text, s.avg_logprob, s.no_speech_prob, s.compression_ratio)
                for s in segments
            ),
        )
```

```python
import argparse
import sys
import time
from datetime import date
from pathlib import Path

from stt.application.dto import CleanRequest, TranscribeRequest
from stt.application.ports import SttError
from stt.application.use_cases import CleanTranscript, TranscribeMedia
from stt.infrastructure.jsonl_store import JsonlStore
from stt.infrastructure.markdown_writer import MarkdownWriter

OUT_ROOT = Path("out")


def _out_dir(source: Path) -> Path:
    if OUT_ROOT.resolve() in source.resolve().parents:
        return source.parent
    return OUT_ROOT / f"{date.today().isoformat()}-{source.stem}"


def _transcribe(args: argparse.Namespace) -> Path:
    from stt.infrastructure.whisper_recognizer import WhisperRecognizer

    def progress(done: float, total: float) -> None:
        percent = 100 * done / total if total else 100
        print(f"\r{percent:5.1f}%  {done:.0f}/{total:.0f} giây", end="", flush=True)

    print(f"Đang chép lời bằng model {args.model}. Lần đầu sẽ tải model về máy.")
    started = time.monotonic()
    result = TranscribeMedia(WhisperRecognizer(args.model), JsonlStore()).execute(
        TranscribeRequest(args.file, _out_dir(args.file), args.language, args.prompt), progress
    )
    elapsed = time.monotonic() - started
    speed = f", gấp {elapsed / result.duration:.1f} lần thời lượng" if result.duration else ""
    print(f"\nĐã chép {result.segment_count} đoạn ({result.language}) trong {elapsed:.0f} giây{speed}.")
    print(f"File thô: {result.raw_path}")
    return result.raw_path


def _clean(raw_path: Path) -> None:
    result = CleanTranscript(JsonlStore(), MarkdownWriter()).execute(CleanRequest(raw_path))
    if not result.complete:
        print("Cảnh báo: lượt chép lời này bị ngắt giữa chừng, transcript chưa đầy đủ.", file=sys.stderr)
    print(f"Giữ {result.kept} đoạn, bỏ {result.dropped} đoạn. Transcript: {result.transcript_path}")


def _build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(prog="stt", description="Chép lời âm thanh máy tính hoặc video, chạy local.")
    commands = parser.add_subparsers(dest="command", required=True)

    def add_source_command(name: str, help_text: str) -> None:
        command = commands.add_parser(name, help=help_text)
        command.add_argument("file", type=Path, help="file wav hoặc video")
        command.add_argument("--model", default="large-v3")
        command.add_argument("--language", choices=["en", "vi"], help="mặc định: tự nhận")
        command.add_argument("--prompt", help="tên riêng, thuật ngữ để model nghe đúng")

    add_source_command("transcribe", "chép lời ra raw.jsonl")
    add_source_command("run", "chép lời rồi làm sạch")
    clean = commands.add_parser("clean", help="làm sạch raw.jsonl thành transcript.md")
    clean.add_argument("raw", type=Path)
    return parser


def main(argv: list[str] | None = None) -> int:
    for stream in (sys.stdout, sys.stderr):
        stream.reconfigure(errors="replace")
    args = _build_parser().parse_args(argv)
    try:
        if args.command == "clean":
            _clean(args.raw)
        else:
            raw_path = _transcribe(args)
            if args.command == "run":
                _clean(raw_path)
    except SttError as error:
        print(f"Lỗi: {error}", file=sys.stderr)
        return 1
    except KeyboardInterrupt:
        print("\nĐã dừng. Phần đã xong vẫn được giữ trên đĩa.", file=sys.stderr)
        return 130
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
class RecordAudio:
    def __init__(self, recorder: AudioRecorder, live_recognizer: SpeechRecognizer | None = None):
        self._recorder = recorder
        self._live_recognizer = live_recognizer

    def execute(
        self, request: RecordRequest, on_text: Callable[[str], None] = lambda text: None
    ) -> RecordResult:
        name = request.name
        if not name or name in (".", "..") or PurePath(name).name != name or "\\" in name:
            raise SttError(f"Tên buổi ghi không hợp lệ: {name!r}. Dùng tên thư mục đơn giản.")
        out_dir = request.out_root / f"{date.today().isoformat()}-{name}"
        audio_path = out_dir / "audio.wav"
        if audio_path.exists():
            raise SttError(f"{audio_path} đã tồn tại. Chọn tên khác để không ghi đè bản thu cũ.")
        out_dir.mkdir(parents=True, exist_ok=True)

        started = time.monotonic()
        chunks = self._recorder.record(audio_path)
        try:
            for chunk in chunks:
                if request.live and self._live_recognizer:
                    recognition = self._live_recognizer.transcribe(io.BytesIO(chunk), None, None)
                    text = " ".join(s.text.strip() for s in recognition.segments)
                    if text:
                        on_text(text)
        except KeyboardInterrupt:
            pass
        finally:
            chunks.close()
        return RecordResult(audio_path, time.monotonic() - started)
```

```python
import io
import threading
import time
import wave
from pathlib import Path
from typing import Iterator

import pyaudiowpatch as pyaudio

from stt.application.ports import SttError

CHUNK_SECONDS = 30


def _wav_bytes(frames: bytes, channels: int, width: int, rate: int) -> bytes:
    buffer = io.BytesIO()
    with wave.open(buffer, "wb") as wav:
        wav.setnchannels(channels)
        wav.setsampwidth(width)
        wav.setframerate(rate)
        wav.writeframes(frames)
    return buffer.getvalue()


def _loopback_device(audio: pyaudio.PyAudio) -> dict:
    try:
        wasapi = audio.get_host_api_info_by_type(pyaudio.paWASAPI)
        speakers = audio.get_device_info_by_index(wasapi["defaultOutputDevice"])
    except OSError as error:
        raise SttError(f"Không truy cập được thiết bị phát âm thanh mặc định: {error}") from error
    if speakers.get("isLoopbackDevice"):
        return speakers
    for device in audio.get_loopback_device_info_generator():
        if speakers["name"] in device["name"]:
            return device
    raise SttError(f"Không tìm thấy thiết bị loopback cho {speakers['name']}.")


class WasapiRecorder:
    def record(self, output_path: Path) -> Iterator[bytes]:
        audio = pyaudio.PyAudio()
        try:
            device = _loopback_device(audio)
            channels = int(device["maxInputChannels"])
            rate = int(device["defaultSampleRate"])
            width = audio.get_sample_size(pyaudio.paInt16)
            chunk_size = CHUNK_SECONDS * rate * channels * width
            lock = threading.Lock()
            pending = bytearray()

            with wave.open(str(output_path), "wb") as wav:
                wav.setnchannels(channels)
                wav.setsampwidth(width)
                wav.setframerate(rate)

                def on_audio(data, frame_count, time_info, status):
                    with lock:
                        wav.writeframes(data)  # cập nhật header mỗi lần, file vẫn mở được nếu chết giữa chừng
                        pending.extend(data)
                    return (None, pyaudio.paContinue)

                try:
                    stream = audio.open(
                        format=pyaudio.paInt16,
                        channels=channels,
                        rate=rate,
                        input=True,
                        input_device_index=device["index"],
                        stream_callback=on_audio,
                    )
                except OSError as error:
                    raise SttError(f"Không mở được luồng thu âm: {error}") from error
                try:
                    while True:
                        time.sleep(0.2)
                        with lock:
                            if len(pending) < chunk_size:
                                continue
                            frames = bytes(pending)
                            pending.clear()
                        yield _wav_bytes(frames, channels, width, rate)
                finally:
                    stream.stop_stream()
                    stream.close()
        finally:
            audio.terminate()
```
